## استفاده

```tsx showLineNumbers
import {
  Popover,
  PopoverContent,
  PopoverDescription,
  PopoverHeader,
  PopoverTitle,
  PopoverTrigger,
} from "@/components/ui/popover"
```

```tsx showLineNumbers
<Popover>
  <PopoverTrigger render={<Button variant="outline" />}>
    باز کردن پاپ‌اور
  </PopoverTrigger>
  <PopoverContent>
    <PopoverHeader>
      <PopoverTitle>عنوان</PopoverTitle>
      <PopoverDescription>متن توضیح اینجا.</PopoverDescription>
    </PopoverHeader>
  </PopoverContent>
</Popover>
```
